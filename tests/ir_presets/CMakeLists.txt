if(NOT TARGET juce::juce_dsp)
    if(NOT JUCE_SOURCE_DIR)
        include(FetchContent)
        FetchContent_Declare(JUCE URL https://github.com/juce-framework/JUCE/archive/refs/tags/8.0.4.zip DOWNLOAD_EXTRACT_TIMESTAMP TRUE)

    else()

    endif()

juce_add_console_app(ir_presets_test PRODUCT_NAME "IR/preset integration")

option(IR_DISCARD_RIGHT_MUTATION "Negative test: build-local discarded right input" OFF)
if(IR_DISCARD_RIGHT_MUTATION)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../Source/PluginProcessor.cpp" mutated_processor)
    string(FIND "${mutated_processor}" "const int ampLanes = numIn >= 2 ? 2 : 1;" mutation_site)
    if(mutation_site EQUAL -1)
        message(FATAL_ERROR "Discard-right mutation site changed; do not silently run unmutated code")
    endif()
    string(REPLACE "const int ampLanes = numIn >= 2 ? 2 : 1;" "const int ampLanes = 1; // TEST MUTATION" mutated_processor "${mutated_processor}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/discard_right.cpp" "${mutated_processor}")
    get_target_property(test_sources ir_presets_test SOURCES)
    list(REMOVE_ITEM test_sources "../../Source/PluginProcessor.cpp")
    set_property(TARGET ir_presets_test PROPERTY SOURCES "${test_sources}")
    target_sources(ir_presets_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/discard_right.cpp")
    set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/discard_right.cpp" APPEND PROPERTY COMPILE_DEFINITIONS VAMP_IR_TEST_CAPTURE=1)
endif()
target_include_directories(ir_presets_test PRIVATE ../../Source)
# Observe the real processor input to the cabinet without duplicating the amp.
set_property(SOURCE ../../Source/PluginProcessor.cpp APPEND PROPERTY COMPILE_DEFINITIONS VAMP_IR_TEST_CAPTURE=1)
target_compile_definitions(ir_presets_test PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1 JucePlugin_Name="Djenty\ Donkey" JucePlugin_VersionString="0.5.6")

include(CTest)

# Test-only API ownership trace, using the real JUCE convolver. No product hook.
option(IR_TRACE_CONVOLUTION "Trace prepare/load/process ownership" ON)
if(IR_TRACE_CONVOLUTION)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    get_target_property(dsp_sources juce_dsp INTERFACE_SOURCES)
    foreach(source IN LISTS dsp_sources)
        if(source MATCHES "/juce_dsp.cpp$")
            get_filename_component(dsp_dir "${source}" DIRECTORY)
            execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/trace_convolution.py" "${dsp_dir}" "${CMAKE_CURRENT_BINARY_DIR}/trace" COMMAND_ERROR_IS_FATAL ANY)
            list(REMOVE_ITEM dsp_sources "${source}")
            list(APPEND dsp_sources "${CMAKE_CURRENT_BINARY_DIR}/trace/traced_dsp.cpp")
            set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/trace/traced_dsp.cpp" APPEND PROPERTY INCLUDE_DIRECTORIES "${dsp_dir}")
        endif()
    endforeach()
    set_property(TARGET juce_dsp PROPERTY INTERFACE_SOURCES "${dsp_sources}")
    set_property(SOURCE ir_presets_test.cpp APPEND PROPERTY COMPILE_DEFINITIONS IR_TRACE_CONVOLUTION=1)
endif()

# Sanitizer configurations instrument all linked JUCE translation units too:
# JUCE Thread uses detached threads plus atomic exit signalling, which a partial
# instrumentation build cannot correctly observe. Never suppress those reports.
set(IR_SANITIZER "" CACHE STRING "Optional thread or address,undefined sanitizer")
if(IR_SANITIZER AND NOT MSVC)
    target_compile_options(ir_presets_test PRIVATE "-fsanitize=${IR_SANITIZER}" -g -O1 -fno-omit-frame-pointer)
    target_link_options(ir_presets_test PRIVATE "-fsanitize=${IR_SANITIZER}")
endif()

    target_link_libraries(compact_vst3_host PRIVATE juce::juce_audio_utils)
endif()
